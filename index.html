-- JUGAAD setup for the exact Storage bucket ID: jugaad_resources -- Run this in Supabase SQL Editor for the SAME project used by the website. -- This does not create or delete the bucket or any uploaded files. -- 1) Diagnostic: this should return exactly one row. select id, name, public, file_size_limit, allowed_mime_types from storage.buckets where id = 'jugaad_resources'; -- 2) Resource metadata policies. alter table public.resources enable row level security; drop policy if exists jugaad_resources_read on public.resources; drop policy if exists jugaad_resources_insert_own on public.resources; drop policy if exists jugaad_resources_delete_own on public.resources; create policy jugaad_resources_read on public.resources for select to authenticated using (true); create policy jugaad_resources_insert_own on public.resources for insert to authenticated with check (uploader_id = (select auth.uid())); create policy jugaad_resources_delete_own on public.resources for delete to authenticated using (uploader_id = (select auth.uid())); grant select, insert, delete on public.resources to authenticated; -- 3) Private Storage bucket policies. -- These replace only policies with these exact names. drop policy if exists jugaad_read_resource_files on storage.objects; drop policy if exists jugaad_upload_own_resource_files on storage.objects; drop policy if exists jugaad_delete_own_resource_files on storage.objects; create policy jugaad_read_resource_files on storage.objects for select to authenticated using (bucket_id = 'jugaad_resources'); create policy jugaad_upload_own_resource_files on storage.objects for insert to authenticated with check ( bucket_id = 'jugaad_resources' and (storage.foldername(name))[1] = (select auth.uid())::text ); create policy jugaad_delete_own_resource_files on storage.objects for delete to authenticated using ( bucket_id = 'jugaad_resources' and (storage.foldername(name))[1] = (select auth.uid())::text );
